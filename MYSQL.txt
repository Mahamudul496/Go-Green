
CREATE TABLE waste_submissions (
    id INT AUTO_INCREMENT PRIMARY KEY,
    waste_type VARCHAR(50),
    weight DECIMAL(10,2),
    details TEXT,
    address TEXT,
    pickup_date DATE,
    time_slot VARCHAR(50),
    image_path VARCHAR(255),
    submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);



CREATE TABLE admin(
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(20),
    email VARCHAR(30),
    password VARCHAR(20)
    
);
